E1-E3. 基础函数参考实现
def is_odd(n):
    return n % 2 != 0

def sum_digits(n):
    total = 0
    for ch in str(abs(n)):
        total += int(ch)
    return total

def is_prime(n):
    if n < 2:
        return False
    i = 2
    while i * i <= n:
        if n % i == 0:
            return False
        i += 1
    return True
E4-E5. 列表分析与字母频次
def analyze_scores(scores):
    if not scores:
        return 0, []
    average = sum(scores) / len(scores)
    above = [x for x in scores if x > average]
    return average, above

def letter_count(text):
    counts = {}
    for ch in text.lower():
        if 'a' <= ch <= 'z':
            counts[ch] = counts.get(ch, 0) + 1
    return counts
E6. 安全密码
def is_safe_password(password):
    if not 8 <= len(password) <= 15:
        return False

    groups = [False, False, False, False]
    special = "@#$%^"
    for ch in password:
        if 'A' <= ch <= 'Z':
            groups[0] = True
        elif 'a' <= ch <= 'z':
            groups[1] = True
        elif '0' <= ch <= '9':
            groups[2] = True
        elif ch in special:
            groups[3] = True
        else:
            return False
    return sum(groups) >= 3
E7. 选择排序
def selection_sort(a, asc=True):
    result = a.copy()
    for i in range(len(result) - 1):
        best = i
        for j in range(i + 1, len(result)):
            if asc and result[j] < result[best]:
                best = j
            elif not asc and result[j] > result[best]:
                best = j
        result[i], result[best] = result[best], result[i]
    return result

每一轮从未排序区选出最小值（升序）或最大值（降序），放到位置 i。

E8. 文件统计
def file_stats(filename):
    line_count = 0
    word_count = 0
    with open(filename, "r", encoding="utf-8") as f:
        for line in f:
            line_count += 1
            word_count += len(line.split())
    return line_count, word_count
