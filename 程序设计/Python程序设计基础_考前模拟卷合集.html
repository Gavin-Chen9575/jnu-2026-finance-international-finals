选择：1-5 B B B A A；6-10 B B B B C；11-15 B B A B B；16-20 C A B C B

# C 应用 2 参考
try:
    total = count = 0
    fails = []
    with open("scores.txt", "r", encoding="utf-8") as f:
        for line in f:
            name, score = line.strip().split(",")
            score = int(score)
            total += score
            count += 1
            if score < 60:
                fails.append(name)
except FileNotFoundError:
    print("scores.txt 不存在")
else:
    print(f"{total/count:.2f}")
    with open("fail.txt", "w", encoding="utf-8") as f:
        for name in fails:
            f.write(name + "\n")
